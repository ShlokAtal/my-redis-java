Short-cut Window 2 Commands

$client = New-Object System.Net.Sockets.TcpClient("localhost",6379)
$stream = $client.GetStream()
$buffer = New-Object byte[] 1024

function Send-RedisCommand($command) {
    $parts = $command -split " "
    $request = "*" + $parts.Count + "`r`n"
    foreach ($part in $parts) {
        $request += '$' + $part.Length + "`r`n" + $part + "`r`n"
    }

    $bytes = [System.Text.Encoding]::ASCII.GetBytes($request)
    $stream.Write($bytes,0,$bytes.Length)

    $count = $stream.Read($buffer,0,$buffer.Length)
    [System.Text.Encoding]::ASCII.GetString($buffer,0,$count)
}
